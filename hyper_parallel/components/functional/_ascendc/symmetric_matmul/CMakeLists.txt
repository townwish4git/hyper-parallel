cmake_minimum_required(VERSION 3.16)
project(hp_symm_pipeline LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(ASCEND_CANN_PACKAGE_PATH "$ENV{ASCEND_HOME_PATH}" CACHE PATH "CANN toolkit")
set(HP_ASCENDC_CMAKE_DIR "${ASCEND_CANN_PACKAGE_PATH}/compiler/tikcpp/ascendc_kernel_cmake")
if(NOT EXISTS "${HP_ASCENDC_CMAKE_DIR}/ascendc.cmake")
    set(HP_ASCENDC_CMAKE_DIR
        "${ASCEND_CANN_PACKAGE_PATH}/${CMAKE_SYSTEM_PROCESSOR}-linux/tikcpp/ascendc_kernel_cmake")
endif()
set(HP_DEVICE_SOURCES fused.cpp)
string(TOLOWER "${SOC_VERSION}" HP_SOC_VERSION)
if(HP_SOC_VERSION MATCHES "^ascend(910b|910_93|950)")
    set(HP_ENABLE_FUSED_NS ON)
    list(APPEND HP_DEVICE_SOURCES fused_ns.cpp)
endif()
if(HP_SOC_VERSION MATCHES "^ascend950")
    # Native ASC compilation preserves the batch scheduling attribute of the mixed kernel.
    set(CMAKE_ASC_ARCHITECTURES dav-3510)
    set(CUSTOM_ASC_CMAKE_PATH "${HP_ASCENDC_CMAKE_DIR}")
    find_package(ASC REQUIRED CONFIG PATHS "${HP_ASCENDC_CMAKE_DIR}" NO_DEFAULT_PATH)
    enable_language(ASC)
    set_source_files_properties(${HP_DEVICE_SOURCES} PROPERTIES LANGUAGE ASC)
    add_library(hp_pipeline_kernel SHARED ${HP_DEVICE_SOURCES})
    target_compile_definitions(hp_pipeline_kernel PRIVATE HP_ASCEND_950)
    target_include_directories(hp_pipeline_kernel PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/..)
else()
    include("${HP_ASCENDC_CMAKE_DIR}/ascendc.cmake")
    ascendc_library(hp_pipeline_kernel SHARED ${HP_DEVICE_SOURCES})
    ascendc_include_directories(hp_pipeline_kernel PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/..)
endif()
add_library(hp_symm_torch SHARED plan.cpp torch_op.cpp)
target_compile_features(hp_symm_torch PRIVATE cxx_std_20)
target_compile_definitions(hp_symm_torch PRIVATE _GLIBCXX_USE_CXX11_ABI=${TORCH_ABI})
if(HP_ENABLE_FUSED_NS)
    target_compile_definitions(hp_symm_torch PRIVATE HP_ENABLE_FUSED_NS)
endif()
target_include_directories(hp_symm_torch PRIVATE
    ${CMAKE_CURRENT_SOURCE_DIR}/.. ${TORCH_DIR}/include ${TORCH_DIR}/include/torch/csrc/api/include
    ${TORCH_NPU_DIR}/include ${PYTHON_INCLUDE} ${ASCEND_CANN_PACKAGE_PATH}/include
    ${ASCEND_CANN_PACKAGE_PATH}/include/experiment ${ASCEND_CANN_PACKAGE_PATH}/asc/include)
target_link_directories(hp_symm_torch PRIVATE ${TORCH_DIR}/lib ${TORCH_NPU_DIR}/lib ${ASCEND_CANN_PACKAGE_PATH}/lib64)
target_link_libraries(hp_symm_torch PRIVATE hp_pipeline_kernel torch_npu torch_cpu c10 tiling_api platform ascendcl)
set_target_properties(hp_symm_torch PROPERTIES BUILD_WITH_INSTALL_RPATH TRUE INSTALL_RPATH "$ORIGIN")
install(TARGETS hp_symm_torch hp_pipeline_kernel LIBRARY DESTINATION . COMPONENT hp_runtime)
